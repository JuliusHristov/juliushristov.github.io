# Parse HTML using Python's standard library so no external dependency is required.
from html.parser import HTMLParser

class ElementRanges(HTMLParser):
    def __init__(self, source):
        super().__init__(convert_charrefs=True)
        self.source = source
        self.starts = []
        self.elements = []
        self.stack = []
        self.lines = [0]
        for i, char in enumerate(source):
            if char == "\n": self.lines.append(i+1)
    def offset(self):
        line, col = self.getpos()
        return self.lines[line-1] + col
    def handle_starttag(self, tag, attrs):
        pos = self.offset()
        if tag in ('br','hr','img','meta','input','link','source','wbr','area','embed'):
            return
        el = dict(tag=tag, start=pos, end=None, parent=self.stack[-1] if self.stack else None)
        self.elements.append(el)
        self.stack.append(el)
    def handle_startendtag(self, tag, attrs): pass
    def handle_endtag(self, tag):
        for i in range(len(self.stack)-1,-1,-1):
            if self.stack[i]['tag'] == tag:
                end = self.source.find('>', self.offset())+1
                for el in self.stack[i:]: el['end'] = end
                del self.stack[i:]
                break
